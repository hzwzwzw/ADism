cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(ADismTests LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  enable_testing()
endif()
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../src)
set(BACKENDS ${SRC}/ui_log.cpp ${SRC}/partition_picker.cpp ${SRC}/sfc_backend.cpp ${SRC}/cbs_backend.cpp ${SRC}/cbs_source_mount.cpp ${SRC}/wim_backend.cpp ${SRC}/boot_backend.cpp ${SRC}/safety.cpp)
add_executable(ADismPolicyTests cbs-policy-test.cpp ${BACKENDS})
add_test(NAME ADismPolicyProtocol COMMAND ADismPolicyTests)
set(TEST_TARGETS ADismPolicyTests)
option(ADISM_BUILD_INTERACTIVE_TESTS "Build UI/partition probes for an explicit lab fixture" OFF)
if(ADISM_BUILD_INTERACTIVE_TESTS)
  add_executable(ADismUiTests ui-test.cpp ${BACKENDS})
  add_executable(ADismPartitionProbe partition-probe.cpp ${SRC}/ui_log.cpp ${SRC}/partition_picker.cpp)
  list(APPEND TEST_TARGETS ADismUiTests ADismPartitionProbe)
endif()
foreach(target IN LISTS TEST_TARGETS)
  target_include_directories(${target} PRIVATE ${SRC})
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WINVER=0x0602 _WIN32_WINNT=0x0602 NOMINMAX)
  target_link_libraries(${target} PRIVATE comctl32 comdlg32 shell32 ole32 oleaut32 advapi32 version uuid wbemuuid uxtheme gdi32 wintrust crypt32)
  if(MINGW)
    target_link_options(${target} PRIVATE -municode -static -static-libgcc -static-libstdc++)
  elseif(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
  endif()
endforeach()
