cmake_minimum_required(VERSION 3.20)
project(ADism VERSION 0.7.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_executable(ADism WIN32 src/gui.cpp src/ui_log.cpp src/partition_picker.cpp src/sfc_backend.cpp src/cbs_backend.cpp src/cbs_source_mount.cpp src/wim_backend.cpp src/boot_backend.cpp src/safety.cpp src/app.rc)
# Resource includes must trigger RC rebuilds in incremental MinGW/Ninja builds.
set_source_files_properties(src/app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
target_compile_definitions(ADism PRIVATE UNICODE _UNICODE WINVER=0x0602 _WIN32_WINNT=0x0602 NOMINMAX)
target_include_directories(ADism PRIVATE src)
target_link_libraries(ADism PRIVATE comctl32 comdlg32 shell32 ole32 oleaut32 advapi32 version uuid wbemuuid uxtheme wintrust crypt32)
if(MINGW)
  target_link_options(ADism PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_compile_options(ADism PRIVATE -Wall -Wextra -Wno-unknown-pragmas)
elseif(MSVC)
  target_compile_options(ADism PRIVATE /utf-8 /W4 /EHsc)
  # app.rc already embeds the administrator/compatibility manifest.
  target_link_options(ADism PRIVATE /MANIFEST:NO)
endif()

include(CTest)
if(BUILD_TESTING)
  add_test(NAME ADismSelfTest COMMAND ADism --self-test)
  add_subdirectory(tests)
endif()
